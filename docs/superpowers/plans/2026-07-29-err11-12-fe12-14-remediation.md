```ts
    /** 직전 저장 시도에서 실패한 행 목록 (성공 시 빈 배열) — ERR-11 */
    const saveFailures = ref<CostSaveFailure[]>([]);
    /** 쓰기는 성공했으나 목록 재조회가 실패한 상태 — ERR-11 */
    const refreshFailedAfterSave = ref(false);

    /**
     * 서버 전송 페이로드를 만든다. 로컬 편집 메타(_status/_localId/_saveError)는 제외한다.
     *
     * @param row 전송할 로컬 행
     * @returns 서버 계약에 맞춘 ItCost 페이로드
     */
    const toPayload = (row: ItCostEx): ItCost => {
        const payload = { ...row, fstDfrDt: toDateString(row.fstDfrDt) } as ItCostEx;
        delete payload._status;
        delete payload._localId;
        delete payload._saveError;
        return payload as ItCost;
    };

    /**
     * 단일 행 쓰기를 수행하고 성공 여부를 반환한다.
     * 실패하면 행에 _saveError를 남기고 saveFailures에 사유를 축적한다(행은 목록에 유지).
     *
     * @param row 대상 행
     * @param operation 쓰기 종류
     * @param write 실제 API 호출
     * @returns 성공 여부
     */
    const runRowWrite = async (
        row: ItCostEx,
        operation: CostSaveFailure['operation'],
        write: () => Promise<unknown>,
    ): Promise<boolean> => {
        try {
            await write();
            delete row._saveError;
            return true;
        } catch (e) {
            console.error(`[CostList] ${operation} 실패 (key=${rowKey(row)}):`, e);
            const message = formatApiError(getErrorMessage(e, '저장에 실패했습니다.'));
            row._saveError = message;
            saveFailures.value = [
                ...saveFailures.value,
                { key: rowKey(row), label: row.cttNm || '이름 없음', operation, message },
            ];
            return false;
        }
    };

    /** 저장 후 목록 재조회 (Task 2에서 실패 처리·재시도 경로를 추가한다) */
    const refreshAfterWrite = async (_successCount: number): Promise<void> => {
        await refreshCostsRaw();
    };
```
